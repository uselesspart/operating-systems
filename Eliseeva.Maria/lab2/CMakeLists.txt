cmake_minimum_required(VERSION 3.13)
project(ipc_chat CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Werror")

option(BUILD_TESTS "Собирать тесты (tests/*.cpp)" ON)
option(REQUIRE_QT "Требовать Qt (если OFF и Qt нет — собираются только тесты)" ON)

find_package(Qt6 COMPONENTS Widgets QUIET)
if(Qt6_FOUND)
    set(QT_LIBS Qt6::Widgets)
    set(HAVE_QT ON)
else()
    find_package(Qt5 COMPONENTS Widgets QUIET)
    if(Qt5_FOUND)
        set(QT_LIBS Qt5::Widgets)
        set(HAVE_QT ON)
    elseif(REQUIRE_QT)
        message(FATAL_ERROR "Qt (Widgets) не найден: установите qt6-base-dev (или qtbase5-dev)")
    else()
        message(WARNING "Qt не найден: собираются только тесты (host_* не будут собраны)")
        set(HAVE_QT OFF)
    endif()
endif()
find_package(Threads REQUIRED)

# Исполняемые файлы host_* кладём в папку проекта
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR})

# Все .cpp в корне подхватываются автоматически (CMakeLists менять не нужно)
file(GLOB ALL_SRCS  CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/*.cpp)
file(GLOB CONN_SRCS CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/conn_*.cpp)
set(COMMON_SRCS ${ALL_SRCS})
list(REMOVE_ITEM COMMON_SRCS ${CONN_SRCS})
# client.cpp нужен только для независимой конфигурации (в варианте 8 клиент — потомок хоста)
list(REMOVE_ITEM COMMON_SRCS ${CMAKE_SOURCE_DIR}/client.cpp)

if(BUILD_TESTS)
    enable_testing()
    set(TEST_OUT ${CMAKE_BINARY_DIR}/tests)
    add_executable(test_core tests/test_core.cpp)
    target_include_directories(test_core PRIVATE ${CMAKE_SOURCE_DIR} ${CMAKE_SOURCE_DIR}/tests)
    target_link_libraries(test_core PRIVATE Threads::Threads)
    set_target_properties(test_core PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${TEST_OUT})
    add_test(NAME core COMMAND test_core)
endif()

if(HAVE_QT)
    add_library(chat_common OBJECT ${COMMON_SRCS})
    target_include_directories(chat_common PUBLIC ${CMAKE_SOURCE_DIR})
    target_link_libraries(chat_common PUBLIC ${QT_LIBS})
endif()

# conn_<код>.cpp  ->  host_<код>  (+ test_conn_<код>)
foreach(conn_src ${CONN_SRCS})
    get_filename_component(conn_name ${conn_src} NAME_WE)
    string(REGEX REPLACE "^conn_" "" type_code ${conn_name})
    if(HAVE_QT)
        add_executable(host_${type_code} $<TARGET_OBJECTS:chat_common> ${conn_src})
        target_include_directories(host_${type_code} PRIVATE ${CMAKE_SOURCE_DIR})
        target_link_libraries(host_${type_code} PRIVATE ${QT_LIBS} Threads::Threads)
        if(EXISTS ${CMAKE_SOURCE_DIR}/client.cpp)
            add_executable(client_${type_code} client.cpp ${conn_src})
            target_link_libraries(client_${type_code} PRIVATE Threads::Threads)
        endif()
    endif()
    if(BUILD_TESTS)
        add_executable(test_conn_${type_code} tests/test_conn.cpp ${conn_src})
        target_include_directories(test_conn_${type_code} PRIVATE ${CMAKE_SOURCE_DIR} ${CMAKE_SOURCE_DIR}/tests)
        target_link_libraries(test_conn_${type_code} PRIVATE Threads::Threads)
        set_target_properties(test_conn_${type_code} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${TEST_OUT})
        add_test(NAME conn_${type_code} COMMAND test_conn_${type_code})
    endif()
endforeach()
