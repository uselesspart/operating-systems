cmake_minimum_required(VERSION 3.16)
project(chat LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Werror)
find_package(Qt6 6.3 REQUIRED COMPONENTS Widgets)
find_package(Threads REQUIRED)

file(GLOB SOURCES CONFIGURE_DEPENDS "*.cpp" "*.c")
file(GLOB CONNECTIONS CONFIGURE_DEPENDS "conn_*.cpp" "conn_*.c")
list(REMOVE_ITEM SOURCES ${CONNECTIONS} "${CMAKE_CURRENT_SOURCE_DIR}/host.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/client.cpp")

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    find_package(Qt6 REQUIRED COMPONENTS Test)
endif()

foreach(CONNECTION IN LISTS CONNECTIONS)
    get_filename_component(TYPE ${CONNECTION} NAME_WE)
    string(REGEX REPLACE "^conn_" "" TYPE ${TYPE})
    add_library(chat_${TYPE} STATIC ${SOURCES} ${CONNECTION})
    target_include_directories(chat_${TYPE} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(chat_${TYPE} PUBLIC TYPE_CODE="${TYPE}")
    target_link_libraries(chat_${TYPE} PUBLIC Qt6::Widgets Threads::Threads)
    foreach(ROLE host client)
        add_executable(${ROLE}_${TYPE} ${ROLE}.cpp)
        target_link_libraries(${ROLE}_${TYPE} PRIVATE chat_${TYPE})
        set_target_properties(${ROLE}_${TYPE} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    endforeach()
    if(BUILD_TESTING)
        file(GLOB TESTS CONFIGURE_DEPENDS "tests/*.cpp")
        foreach(TEST IN LISTS TESTS)
            get_filename_component(NAME ${TEST} NAME_WE)
            add_executable(${NAME}_${TYPE} ${TEST})
            target_link_libraries(${NAME}_${TYPE} PRIVATE chat_${TYPE} Qt6::Test)
            set_target_properties(${NAME}_${TYPE} PROPERTIES AUTOMOC ON)
            if(NOT NAME STREQUAL "driver")
                add_test(NAME ${NAME}_${TYPE} COMMAND ${NAME}_${TYPE})
                set_tests_properties(${NAME}_${TYPE} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
            endif()
        endforeach()
        add_test(NAME integration_${TYPE}
            COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py" $<TARGET_FILE:driver_${TYPE}>)
        set_tests_properties(integration_${TYPE} PROPERTIES TIMEOUT 180)
    endif()
endforeach()
