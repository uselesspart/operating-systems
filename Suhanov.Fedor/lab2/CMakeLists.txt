cmake_minimum_required(VERSION 3.24)
project(PolyNet VERSION 1.0 LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "PolyNet requires Linux kernel 5.3 or newer.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(Threads REQUIRED)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)

# New implementation files and conn_<code>.cpp transports are discovered on make.
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "src/*.cpp" "src/*.cc" "src/*.c")
file(GLOB TRANSPORTS CONFIGURE_DEPENDS "src/conn_*.cpp")
list(REMOVE_ITEM SOURCES ${TRANSPORTS} "${CMAKE_CURRENT_SOURCE_DIR}/src/host.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/client.cpp")
foreach(TRANSPORT IN LISTS TRANSPORTS)
    get_filename_component(CODE "${TRANSPORT}" NAME_WE)
    string(REGEX REPLACE "^conn_" "" CODE "${CODE}")
    add_library(core_${CODE} STATIC ${SOURCES} "${TRANSPORT}")
    target_include_directories(core_${CODE} PUBLIC include)
    target_compile_options(core_${CODE} PRIVATE -Wall -Wextra -Werror)
    target_compile_definitions(core_${CODE} PUBLIC TYPE_CODE="${CODE}")
    target_link_libraries(core_${CODE} PUBLIC Threads::Threads rt Qt${QT_VERSION_MAJOR}::Widgets)
    foreach(ROLE host client)
        add_executable(${ROLE}_${CODE} "src/${ROLE}.cpp")
        target_compile_options(${ROLE}_${CODE} PRIVATE -Wall -Wextra -Werror)
        target_link_libraries(${ROLE}_${CODE} PRIVATE core_${CODE})
    endforeach()
endforeach()

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(NOT Python3_FOUND)
        set(Python3_EXECUTABLE python3)
    endif()
    foreach(TRANSPORT IN LISTS TRANSPORTS)
        get_filename_component(CODE "${TRANSPORT}" NAME_WE)
        string(REGEX REPLACE "^conn_" "" CODE "${CODE}")
        add_test(NAME chat_${CODE} COMMAND xvfb-run -a ${Python3_EXECUTABLE}
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py"
            --bin-dir "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}" --transport "${CODE}")
        set_tests_properties(chat_${CODE} PROPERTIES TIMEOUT 120)
    endforeach()
endif()
