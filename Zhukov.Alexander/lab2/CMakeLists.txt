cmake_minimum_required(VERSION 3.16)
project(os_chat LANGUAGES C CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Build on Linux")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_compile_options(-Wall -Werror)

find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(Threads REQUIRED)
option(BUILD_TESTING "Build tests" OFF)

file(GLOB_RECURSE sources CONFIGURE_DEPENDS src/*.cpp src/*.c)
file(GLOB_RECURSE headers CONFIGURE_DEPENDS include/*.h)
set(common_sources)
set(conn_sources)

foreach(source IN LISTS sources)
    get_filename_component(name "${source}" NAME_WE)
    if(name STREQUAL "host" OR name STREQUAL "client")
        continue()
    elseif(name MATCHES "^conn_(.+)$")
        list(APPEND conn_sources "${source}")
    else()
        list(APPEND common_sources "${source}")
    endif()
endforeach()

add_library(chat_common STATIC ${common_sources} ${headers})
target_include_directories(chat_common PUBLIC include)
target_link_libraries(chat_common PUBLIC Qt6::Widgets Threads::Threads rt)

foreach(source IN LISTS conn_sources)
    get_filename_component(name "${source}" NAME_WE)
    string(REGEX REPLACE "^conn_" "" type "${name}")
    add_library("${name}" OBJECT "${source}")
    target_link_libraries("${name}" PRIVATE chat_common)

    foreach(role host client)
        add_executable("${role}_${type}" "src/${role}.cpp")
        target_link_libraries("${role}_${type}" PRIVATE chat_common "${name}")
    endforeach()
endforeach()

if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    file(GLOB_RECURSE test_sources CONFIGURE_DEPENDS tests/*.cpp tests/*.c)
    foreach(source IN LISTS conn_sources)
        get_filename_component(name "${source}" NAME_WE)
        string(REGEX REPLACE "^conn_" "" type "${name}")
        add_executable("run_tests_${type}" ${test_sources})
        target_link_libraries("run_tests_${type}" PRIVATE chat_common Qt6::Test "${name}")
    endforeach()
endif()
