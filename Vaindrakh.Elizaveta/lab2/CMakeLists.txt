cmake_minimum_required(VERSION 3.16)
project(wolf_and_goat LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Werror)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This lab uses Linux IPC. Build on Linux or in WSL.")
endif()
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets Test)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets Test)
find_package(Threads REQUIRED)
include(CTest)

# Discover every source and transport automatically, including newly added files.
file(GLOB_RECURSE ALL_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.c")
file(GLOB TRANSPORTS CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/conn_*.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/conn_*.c")
# Transport classes use the C++ interface even when the source has a .c suffix.
set_source_files_properties(${TRANSPORTS} PROPERTIES LANGUAGE CXX)
set(COMMON_SOURCES ${ALL_SOURCES})
list(REMOVE_ITEM COMMON_SOURCES ${TRANSPORTS}
    "${CMAKE_CURRENT_SOURCE_DIR}/src/host.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/session.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/window.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/options.cpp")
add_library(lab_common OBJECT ${COMMON_SOURCES})
target_include_directories(lab_common PUBLIC include)
target_link_libraries(lab_common PUBLIC Qt${QT_VERSION_MAJOR}::Widgets Threads::Threads)
foreach(TRANSPORT IN LISTS TRANSPORTS)
    get_filename_component(NAME "${TRANSPORT}" NAME_WE)
    string(REGEX REPLACE "^conn_" "" TYPE "${NAME}")
    add_library(backend_${TYPE} OBJECT "${TRANSPORT}" src/options.cpp)
    target_include_directories(backend_${TYPE} PUBLIC include)
    target_compile_definitions(backend_${TYPE} PUBLIC TYPE_CODE="${TYPE}")
    target_link_libraries(backend_${TYPE} PUBLIC Threads::Threads)
    # Common source files which print the transport are compiled per transport.
    add_executable(host_${TYPE} src/host.cpp src/session.cpp src/window.cpp)
    target_link_libraries(host_${TYPE} PRIVATE backend_${TYPE} lab_common Qt${QT_VERSION_MAJOR}::Widgets Threads::Threads rt)
    set_target_properties(host_${TYPE} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    install(TARGETS host_${TYPE} RUNTIME DESTINATION bin)
    if(BUILD_TESTING)
        add_executable(test_ipc_${TYPE} tests/test_ipc.cpp)
        target_link_libraries(test_ipc_${TYPE} PRIVATE backend_${TYPE} lab_common Threads::Threads rt)
        add_test(NAME ipc_${TYPE} COMMAND test_ipc_${TYPE})
        set_tests_properties(ipc_${TYPE} PROPERTIES TIMEOUT 20)
        add_executable(test_gui_${TYPE} tests/test_gui.cpp src/session.cpp src/window.cpp)
        target_link_libraries(test_gui_${TYPE} PRIVATE backend_${TYPE} lab_common Qt${QT_VERSION_MAJOR}::Test rt)
        add_test(NAME gui_${TYPE} COMMAND test_gui_${TYPE})
        set_tests_properties(gui_${TYPE} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
    endif()
endforeach()
if(BUILD_TESTING)
    add_executable(test_game tests/test_game.cpp)
    target_link_libraries(test_game PRIVATE lab_common)
    add_test(NAME game COMMAND test_game)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_integration.py" "${CMAKE_BINARY_DIR}/bin")
    set_tests_properties(integration PROPERTIES TIMEOUT 100)
endif()
