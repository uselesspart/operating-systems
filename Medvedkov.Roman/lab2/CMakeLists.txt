cmake_minimum_required(VERSION 3.16)
project(RomanLocalChat VERSION 1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_compile_options(-Wall -Wextra -Werror)
add_compile_definitions(_FORTIFY_SOURCE=2)
find_package(Qt5 5.12 REQUIRED COMPONENTS Widgets Test)
find_package(Threads REQUIRED)
include(CTest)

# Ordinary sources and conn_TYPE modules are discovered on every configuration.
file(GLOB_RECURSE IMPLEMENTATIONS CONFIGURE_DEPENDS src/*.cpp src/*.c)
file(GLOB ROOT_IMPLEMENTATIONS CONFIGURE_DEPENDS *.cpp *.c)
list(APPEND IMPLEMENTATIONS ${ROOT_IMPLEMENTATIONS})
set(TRANSPORTS ${IMPLEMENTATIONS})
list(FILTER TRANSPORTS INCLUDE REGEX "/conn_[^/]+\.(cpp|c)$")
list(REMOVE_ITEM IMPLEMENTATIONS ${TRANSPORTS} "${PROJECT_SOURCE_DIR}/src/host.cpp")
if(NOT TRANSPORTS)
    message(FATAL_ERROR "No src/conn_TYPE.cpp transport implementations found")
endif()
add_library(chat_core STATIC ${IMPLEMENTATIONS})
target_include_directories(chat_core PUBLIC include)
target_link_libraries(chat_core PUBLIC Qt5::Widgets Threads::Threads rt)

if(BUILD_TESTING)
    add_executable(test_model tests/test_model.cpp)
    target_link_libraries(test_model PRIVATE chat_core)
    add_test(NAME model COMMAND test_model)
    file(GLOB SCENARIOS CONFIGURE_DEPENDS tests/test_*.cpp)
    list(REMOVE_ITEM SCENARIOS "${PROJECT_SOURCE_DIR}/tests/test_model.cpp")
endif()

foreach(transport IN LISTS TRANSPORTS)
    get_filename_component(code "${transport}" NAME_WE)
    string(REGEX REPLACE "^conn_" "" code "${code}")
    add_executable(host_${code} src/host.cpp "${transport}")
    target_link_libraries(host_${code} PRIVATE chat_core)
    set_target_properties(host_${code} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")
    if(BUILD_TESTING)
        foreach(scenario IN LISTS SCENARIOS)
            get_filename_component(test_name "${scenario}" NAME_WE)
            add_executable(${test_name}_${code} "${scenario}" "${transport}")
            target_link_libraries(${test_name}_${code} PRIVATE chat_core Qt5::Test)
            add_test(NAME ${test_name}_${code} COMMAND ${test_name}_${code})
            set_tests_properties(${test_name}_${code} PROPERTIES TIMEOUT 40 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        endforeach()
    endif()
endforeach()
