cmake_minimum_required(VERSION 3.16)
project(lab2_chat LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)

file(GLOB transports CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/conn_*.cpp")
file(GLOB shared CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
list(REMOVE_ITEM shared "${CMAKE_CURRENT_SOURCE_DIR}/host.cpp" ${transports})

include(CTest)
foreach(transport IN LISTS transports)
    get_filename_component(stem "${transport}" NAME_WE)
    string(REGEX REPLACE "^conn_" "" code "${stem}")
    add_library("chat_${code}" OBJECT ${shared} "${transport}")
    target_compile_options("chat_${code}" PRIVATE -Wall -Werror)
    target_link_libraries("chat_${code}" PUBLIC Threads::Threads rt)
    if(code STREQUAL "sock")
        target_compile_definitions("chat_${code}" PRIVATE CHAT_SOCKET)
    endif()

    add_executable("host_${code}" host.cpp)
    target_compile_options("host_${code}" PRIVATE -Wall -Werror)
    target_compile_definitions("host_${code}" PRIVATE CHAT_TRANSPORT="${code}")
    target_link_libraries("host_${code}" PRIVATE "chat_${code}" Qt${QT_VERSION_MAJOR}::Widgets)

    if(BUILD_TESTING)
        add_executable("test_${code}" tests/exchange.cpp)
        target_include_directories("test_${code}" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        target_compile_options("test_${code}" PRIVATE -Wall -Werror)
        target_link_libraries("test_${code}" PRIVATE "chat_${code}")
        add_test(NAME "exchange_${code}" COMMAND "test_${code}")
        set_tests_properties("exchange_${code}" PROPERTIES TIMEOUT 20)
    endif()
endforeach()
